cmake_minimum_required(VERSION 3.16)

project(cmake-toolset-rpath-test LANGUAGES C CXX)

set(ATFRAMEWORK_CMAKE_TOOLSET_DIR "${CMAKE_CURRENT_LIST_DIR}/../..")
set(PROJECT_THIRD_PARTY_PACKAGE_DIR "${CMAKE_CURRENT_BINARY_DIR}/packages")
set(PROJECT_THIRD_PARTY_INSTALL_DIR "${CMAKE_CURRENT_BINARY_DIR}/target-prefix")
set(PROJECT_THIRD_PARTY_HOST_INSTALL_DIR "${CMAKE_CURRENT_BINARY_DIR}/host-prefix")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/lib")
set(CMAKE_TOOLSET_RPATH_TEST_TARGET_USER_RPATH "${CMAKE_CURRENT_BINARY_DIR}/target-user-rpath")
set(CMAKE_TOOLSET_RPATH_TEST_HOST_USER_RPATH "${CMAKE_CURRENT_BINARY_DIR}/host-user-rpath")
set(CMAKE_BUILD_RPATH
    "${CMAKE_TOOLSET_RPATH_TEST_TARGET_USER_RPATH}"
    CACHE STRING "Target build RPATH fixture" FORCE)
set(CMAKE_HOST_BUILD_RPATH
    "${CMAKE_TOOLSET_RPATH_TEST_HOST_USER_RPATH}"
    CACHE STRING "Host build RPATH fixture" FORCE)

set(ENV{LD_LIBRARY_PATH} "cmake-toolset-rpath-sentinel")
include("${ATFRAMEWORK_CMAKE_TOOLSET_DIR}/Import.cmake")

if(APPLE)
  set(CMAKE_TOOLSET_RPATH_TEST_TARGET_ORIGIN "@loader_path")
elseif(UNIX)
  set(CMAKE_TOOLSET_RPATH_TEST_TARGET_ORIGIN "$ORIGIN")
endif()
if(CMAKE_HOST_APPLE)
  set(CMAKE_TOOLSET_RPATH_TEST_HOST_ORIGIN "@loader_path")
elseif(CMAKE_HOST_UNIX)
  set(CMAKE_TOOLSET_RPATH_TEST_HOST_ORIGIN "$ORIGIN")
endif()

set(CMAKE_TOOLSET_RPATH_TEST_TARGET_PREFIX_RPATH "${PROJECT_THIRD_PARTY_INSTALL_DIR}/${CMAKE_INSTALL_LIBDIR}")
set(CMAKE_TOOLSET_RPATH_TEST_HOST_PREFIX_RPATH "${PROJECT_THIRD_PARTY_HOST_INSTALL_DIR}/${CMAKE_INSTALL_LIBDIR}")
set(CMAKE_TOOLSET_RPATH_TEST_TARGET_EXPECTED_RPATH "${CMAKE_TOOLSET_RPATH_TEST_TARGET_PREFIX_RPATH}")
set(CMAKE_TOOLSET_RPATH_TEST_HOST_EXPECTED_RPATH "${CMAKE_TOOLSET_RPATH_TEST_HOST_PREFIX_RPATH}")
if(NOT ATFRAMEWORK_CMAKE_TOOLSET_THIRD_PARTY_USE_ABSOLUTE_RPATH AND CMAKE_TOOLSET_RPATH_TEST_TARGET_ORIGIN)
  file(RELATIVE_PATH CMAKE_TOOLSET_RPATH_TEST_RELATIVE_DIR "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}"
       "${CMAKE_TOOLSET_RPATH_TEST_TARGET_PREFIX_RPATH}")
  set(CMAKE_TOOLSET_RPATH_TEST_TARGET_EXPECTED_RPATH
      "${CMAKE_TOOLSET_RPATH_TEST_TARGET_ORIGIN}/${CMAKE_TOOLSET_RPATH_TEST_RELATIVE_DIR}")
endif()
if(NOT ATFRAMEWORK_CMAKE_TOOLSET_THIRD_PARTY_USE_ABSOLUTE_RPATH AND CMAKE_TOOLSET_RPATH_TEST_HOST_ORIGIN)
  file(RELATIVE_PATH CMAKE_TOOLSET_RPATH_TEST_RELATIVE_DIR "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}"
       "${CMAKE_TOOLSET_RPATH_TEST_HOST_PREFIX_RPATH}")
  set(CMAKE_TOOLSET_RPATH_TEST_HOST_EXPECTED_RPATH
      "${CMAKE_TOOLSET_RPATH_TEST_HOST_ORIGIN}/${CMAKE_TOOLSET_RPATH_TEST_RELATIVE_DIR}")
endif()

foreach(cmake_toolset_rpath_test_var CMAKE_BUILD_RPATH COMPILER_OPTION_INHERIT_CMAKE_BUILD_RPATH)
  if(NOT CMAKE_TOOLSET_RPATH_TEST_TARGET_EXPECTED_RPATH IN_LIST ${cmake_toolset_rpath_test_var}
     OR NOT CMAKE_TOOLSET_RPATH_TEST_TARGET_USER_RPATH IN_LIST ${cmake_toolset_rpath_test_var})
    message(FATAL_ERROR "${cmake_toolset_rpath_test_var} does not contain the expected target RPATH entries")
  endif()
endforeach()
foreach(cmake_toolset_rpath_test_var CMAKE_HOST_BUILD_RPATH COMPILER_OPTION_INHERIT_CMAKE_HOST_BUILD_RPATH)
  if(NOT CMAKE_TOOLSET_RPATH_TEST_HOST_EXPECTED_RPATH IN_LIST ${cmake_toolset_rpath_test_var}
     OR NOT CMAKE_TOOLSET_RPATH_TEST_HOST_USER_RPATH IN_LIST ${cmake_toolset_rpath_test_var})
    message(FATAL_ERROR "${cmake_toolset_rpath_test_var} does not contain the expected host RPATH entries")
  endif()
endforeach()
if(NOT "${CMAKE_BUILD_RPATH}" STREQUAL "${COMPILER_OPTION_INHERIT_CMAKE_BUILD_RPATH}"
   OR NOT "${CMAKE_HOST_BUILD_RPATH}" STREQUAL "${COMPILER_OPTION_INHERIT_CMAKE_HOST_BUILD_RPATH}")
  message(FATAL_ERROR "Build RPATH is not inherited unchanged")
endif()

if(DEFINED CMAKE_TOOLSET_RPATH_TEST_EXPECT_USE_ORIGIN)
  if(NOT "${CMAKE_BUILD_RPATH_USE_ORIGIN}" STREQUAL "${CMAKE_TOOLSET_RPATH_TEST_EXPECT_USE_ORIGIN}"
     OR NOT "${CMAKE_HOST_BUILD_RPATH_USE_ORIGIN}" STREQUAL "${CMAKE_TOOLSET_RPATH_TEST_EXPECT_USE_ORIGIN}"
     OR NOT "${COMPILER_OPTION_INHERIT_CMAKE_BUILD_RPATH_USE_ORIGIN}" STREQUAL
        "${CMAKE_TOOLSET_RPATH_TEST_EXPECT_USE_ORIGIN}"
     OR NOT "${COMPILER_OPTION_INHERIT_CMAKE_HOST_BUILD_RPATH_USE_ORIGIN}" STREQUAL
        "${CMAKE_TOOLSET_RPATH_TEST_EXPECT_USE_ORIGIN}")
    message(FATAL_ERROR "Outer BUILD_RPATH_USE_ORIGIN settings were changed")
  endif()
elseif(ATFRAMEWORK_CMAKE_TOOLSET_THIRD_PARTY_USE_ABSOLUTE_RPATH)
  if(CMAKE_BUILD_RPATH_USE_ORIGIN OR CMAKE_HOST_BUILD_RPATH_USE_ORIGIN)
    message(FATAL_ERROR "Absolute RPATH mode unexpectedly enables origin mode")
  endif()
elseif(NOT CMAKE_BUILD_RPATH_USE_ORIGIN OR NOT CMAKE_HOST_BUILD_RPATH_USE_ORIGIN)
  message(FATAL_ERROR "Relative RPATH mode did not enable origin mode")
elseif(CMAKE_TOOLSET_RPATH_TEST_TARGET_ORIGIN AND CMAKE_TOOLSET_RPATH_TEST_TARGET_PREFIX_RPATH IN_LIST
                                                  CMAKE_BUILD_RPATH)
  message(FATAL_ERROR "Target RPATH is not loader-relative")
elseif(CMAKE_TOOLSET_RPATH_TEST_HOST_ORIGIN AND CMAKE_TOOLSET_RPATH_TEST_HOST_PREFIX_RPATH IN_LIST
                                                CMAKE_HOST_BUILD_RPATH)
  message(FATAL_ERROR "Host RPATH is not loader-relative")
endif()
if(NOT "$ENV{LD_LIBRARY_PATH}" STREQUAL "cmake-toolset-rpath-sentinel")
  message(FATAL_ERROR "Import.cmake changed LD_LIBRARY_PATH")
endif()

file(
  GENERATE
  OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/rpath-test.c"
  CONTENT "#if defined(_WIN32)\n__declspec(dllexport)\n#endif\nint cmake_toolset_rpath_test(void) { return 0; }\n")
add_library(cmake-toolset-rpath-test SHARED "${CMAKE_CURRENT_BINARY_DIR}/rpath-test.c")
set_target_properties(
  cmake-toolset-rpath-test PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_TOOLSET_RPATH_TEST_TARGET_PREFIX_RPATH}"
                                      RUNTIME_OUTPUT_DIRECTORY "${PROJECT_THIRD_PARTY_INSTALL_DIR}/bin")

file(
  GENERATE
  OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/rpath-test-main.c"
  CONTENT "int cmake_toolset_rpath_test(void);\nint main(void) { return cmake_toolset_rpath_test(); }\n")
add_executable(cmake-toolset-rpath-run "${CMAKE_CURRENT_BINARY_DIR}/rpath-test-main.c")
target_link_libraries(cmake-toolset-rpath-run PRIVATE cmake-toolset-rpath-test)
